cmake_minimum_required(VERSION 3.8)
project(lwr_hw)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
set(hw_dependencies hardware_interface rclcpp rclcpp_lifecycle pluginlib urdf kdl_parser
  orocos_kdl_vendor joint_limits control_toolbox std_msgs)
foreach(dependency IN LISTS hw_dependencies)
  find_package(${dependency} REQUIRED)
endforeach()
find_package(gz_ros2_control REQUIRED)
find_package(ignition-gazebo6 REQUIRED)
find_package(angles REQUIRED)
find_package(Threads REQUIRED)
add_library(friremote SHARED src/fri/friremote.cpp src/fri/friudp.cpp)
target_compile_definitions(friremote PRIVATE HAVE_SOCKLEN_T)
target_include_directories(friremote PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
add_library(lwr_hw SHARED src/lwr_hw.cpp)
target_include_directories(lwr_hw PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(lwr_hw ${hw_dependencies})
add_library(lwr_hw_fri SHARED src/lwr_hw_fri_node.cpp)
ament_target_dependencies(lwr_hw_fri ${hw_dependencies})
target_link_libraries(lwr_hw_fri lwr_hw friremote Threads::Threads)
pluginlib_export_plugin_description_file(hardware_interface lwr_hw_plugins.xml)
add_library(lwr_hw_gazebo_plugin SHARED src/lwr_hw_gazebo_plugin.cpp)
ament_target_dependencies(lwr_hw_gazebo_plugin ${hw_dependencies} gz_ros2_control angles)
target_link_libraries(lwr_hw_gazebo_plugin lwr_hw ignition-gazebo6::core)
pluginlib_export_plugin_description_file(gz_ros2_control lwr_hw_gazebo_plugins.xml)
# Same external FRIL backend as ROS 1; never download or replace it implicitly.
option(LWR_HW_WITH_FRIL "Build the original external Stanford FRIL backend" OFF)
if(LWR_HW_WITH_FRIL)
  find_path(FRIL_INCLUDE_DIR FastResearchInterface.h PATH_SUFFIXES fril REQUIRED)
  find_library(FRIL_LIBRARY NAMES fril REQUIRED)
  add_library(lwr_hw_fril SHARED src/lwr_hw_fril_node.cpp)
  target_include_directories(lwr_hw_fril PRIVATE ${FRIL_INCLUDE_DIR})
  ament_target_dependencies(lwr_hw_fril ${hw_dependencies})
  target_link_libraries(lwr_hw_fril lwr_hw ${FRIL_LIBRARY})
  pluginlib_export_plugin_description_file(hardware_interface lwr_hw_fril_plugins.xml)
  install(TARGETS lwr_hw_fril LIBRARY DESTINATION lib)
endif()
install(TARGETS lwr_hw lwr_hw_fri friremote EXPORT export_lwr_hw
  ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
# Gazebo plugin library: loaded by gz_ros2_control, not linked by other packages
install(TARGETS lwr_hw_gazebo_plugin LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY launch krl doc DESTINATION share/${PROJECT_NAME} PATTERN "__pycache__" EXCLUDE)
install(PROGRAMS scripts/fri_monitor.py DESTINATION lib/${PROJECT_NAME})
ament_export_targets(export_lwr_hw HAS_LIBRARY_TARGET)
ament_export_dependencies(${hw_dependencies} Threads)
ament_package()
