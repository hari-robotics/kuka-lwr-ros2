cmake_minimum_required(VERSION 3.8)
project(cartesian_position_node)
find_package(ament_cmake REQUIRED)
foreach(dependency rclcpp sensor_msgs std_msgs geometry_msgs lwr_controllers kdl_parser orocos_kdl_vendor)
  find_package(${dependency} REQUIRED)
endforeach()
add_executable(cartesian_position_node src/cartesian_position_node.cpp)
target_compile_features(cartesian_position_node PUBLIC cxx_std_17)
target_compile_options(cartesian_position_node PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(cartesian_position_node
  rclcpp sensor_msgs std_msgs geometry_msgs lwr_controllers kdl_parser orocos_kdl_vendor)
install(TARGETS cartesian_position_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
# Tests live in test/ and stay out of the default (user) build and install space.
# Build them explicitly: colcon build --cmake-args -DLWR_BUILD_TESTS=ON
option(LWR_BUILD_TESTS "Build the test suite in test/ (also requires BUILD_TESTING)" OFF)
if(BUILD_TESTING AND LWR_BUILD_TESTS)
  add_subdirectory(test)
endif()
ament_package()
